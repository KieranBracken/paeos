def _canonical(value: object) -> bytes:
    try:
        return json.dumps(value, sort_keys=True, separators=(",", ":"),
                          ensure_ascii=True, allow_nan=False).encode("ascii")
    except (ValueError, TypeError) as exc:
        raise SupersessionError("value is not canonical JSON") from exc


def request_digest(request: SupersessionRequest) -> str:
    """Exact public signing input; hashing a request does not authorize it."""
    if type(request) is not SupersessionRequest:
        raise SupersessionError("exact SupersessionRequest required")
    return content_hash(_canonical(asdict(request)))


def _identifier(value: object, label: str) -> str:
    if not isinstance(value, str) or not _IDENTIFIER.fullmatch(value):
        raise SupersessionError(f"invalid {label}")
    return value


def _hash(value: object, label: str) -> str:
    if not isinstance(value, str) or not is_valid_hash(value):
        raise SupersessionError(f"invalid SHA256 reference: {label}")
    return value


def _path(value: object) -> str:
    if (not isinstance(value, str) or not value or value != value.strip()
        or any(ord(char) < 32 for char in value)
        or value.startswith("/") or "\\" in value
        or any(char in value for char in "*?[]:")
        or any(part in ("", ".", "..") for part in value.split("/"))
        or posixpath.normpath(value) != value):
        raise SupersessionError("scope requires exact canonical relative file paths")
    return value


def _strings(value: object, label: str, *, paths: bool = False) -> tuple[str, ...]:
    if not isinstance(value, list) or any(not isinstance(item, str) for item in value):
        raise SupersessionError(f"{label} must be a string array")
    if len(value) != len(set(value)):
        raise SupersessionError(f"{label} contains duplicates")
    for item in value:
        if paths:
            _path(item)
        elif not item:
            raise SupersessionError(f"{label} contains an empty value")
    return tuple(value)


def _object(value: object, fields: set[str], label: str) -> dict:
    if not isinstance(value, dict) or set(value) != fields:
        raise SupersessionError(f"{label} has missing or unknown fields")
    return value

    def __init__(self, *, cas: CAS, ledger: Ledger, verifier: CapabilityVerifier,
                 goal_id: str, run_id: str, session: str, domain: DomainTag,
                 policy_ref: str) -> None:
        self._cas = cas
        self._ledger = ledger
        self._verifier = verifier
        self._goal_id = _identifier(goal_id, "goal")
        self._run_id = _identifier(run_id, "run")
        self._session = _identifier(session, "session")
        if not isinstance(domain, DomainTag):
            raise SupersessionError("explicit execution domain required")
        self._domain = domain.value
        self._policy_ref = _hash(policy_ref, "trusted policy")

    def _read(self, ref: str) -> bytes:
        _hash(ref, "CAS")
        raw = self._cas.get(ref)
        # Enforce integrity even when the caller's CAS disables verify_on_read.
        if content_hash(raw) != ref or not raw:
            raise SupersessionError("CAS content is empty or does not match its reference")
        return raw

    def _json(self, ref: str) -> dict:
        raw = self._read(ref)
        def pairs(items):
            result = {}
            for key, value in items:
                if key in result:
                    raise SupersessionError("duplicate JSON member")
                result[key] = value
            return result
        try:
            value = json.loads(raw, object_pairs_hook=pairs)
        except (ValueError, UnicodeError) as exc:
            raise SupersessionError("invalid JSON artifact") from exc
        if not isinstance(value, dict) or _canonical(value) != raw:
            raise SupersessionError("artifact must be a canonically encoded JSON object")
        return value

    def _obligations(self, policy: dict, scope: set[str]) -> tuple[tuple[str, ...], tuple[str, ...]]:
        _object(policy, {"schema", "authorized_paths", "protected_paths", "claims"}, "policy")
        if policy["schema"] != "paeos.scope-policy/1":
            raise SupersessionError("unknown policy schema")
        allowed = set(_strings(policy["authorized_paths"], "authorized paths", paths=True))
        protected = set(_strings(policy["protected_paths"], "protected paths", paths=True))
        if not scope <= allowed:
            raise SupersessionError("successor scope exceeds the installed exact policy")
        claims = policy["claims"]
        if not isinstance(claims, list):
            raise SupersessionError("claims must be an array")
        graph = {}
        for claim in claims:
            _object(claim, {"id", "source_paths", "depends_on", "evidence_refs"}, "claim")
            claim_id = _identifier(claim["id"], "claim id")
            if claim_id in graph:
                raise SupersessionError("duplicate claim id")
            sources = _strings(claim["source_paths"], "claim sources", paths=True)
            dependencies = _strings(claim["depends_on"], "claim dependencies")
            evidence = _strings(claim["evidence_refs"], "claim evidence")
            for ref in evidence:
                self._read(ref)
            graph[claim_id] = (set(sources), set(dependencies), set(evidence))
        if any(dependencies - graph.keys() for _, dependencies, _ in graph.values()):
            raise SupersessionError("dangling baseline dependency")
        # Kahn's algorithm validates the graph without recursion limits or implicit claims.
        remaining = set(graph)
        while remaining:
            ready = {name for name in remaining if not (graph[name][1] & remaining)}
            if not ready:
                raise SupersessionError("cyclic baseline dependencies")
            remaining -= ready
        impacted = {name for name, (sources, _, _) in graph.items() if sources & scope}
        while True:
            expanded = impacted | {name for name, (_, dependencies, _) in graph.items()
                                   if dependencies & impacted}
            if expanded == impacted:
                break
            impacted = expanded
        if scope & protected and "P1" not in impacted:
            raise SupersessionError("protected overlap requires mechanically impacted P1")
        stale = {ref for name in impacted for ref in graph[name][2]}

    def consume(self, request: SupersessionRequest, authority: CapabilityToken) -> ConsumptionReceipt:
        """Consume once, retaining pending obligations. Never return execution authority.

        An unclassified exception from the atomic append may be a committed operation
        whose response was lost. It propagates unchanged; there is no retry here.
        """
        try:
            atomic_append = getattr(self._ledger, "append_once_if_new", None)
            if not callable(atomic_append):
                raise SupersessionError("atomic append_once_if_new ledger port is required")
            digest = request_digest(request)
            if type(authority) is not CapabilityToken:
                raise SupersessionError("existing kernel capability is required; evidence is not authority")
            obligations, stale = self._validate_inputs(request)
            head_hash = self._ledger.verify_chain()
            head = self._ledger.head()
            if head_hash != request.expected_head_hash or (head and head.row_hash != head_hash):
                raise SupersessionError("stale ledger predecessor")
            next_seq = (head.seq if head else 0) + 1
            if next_seq > request.expiry_seq:
                raise SupersessionError("request sequence expiry exceeded")
            self._verifier.verify(
                authority, goal_id=self._goal_id, run_id=self._run_id, stage=StageId.PLAN,
                role=Role.RATIFIER, session=self._session, current_seq=next_seq,
                operation="scope.supersede:" + digest, mission_execution_digest=digest,
            )

            request_ref = self._cas.put(_canonical(asdict(request)))
            binding = authority.bound_to
            token_ref = self._cas.put(_canonical({
                "token": authority.token,
                "bound_to": {"goal_id": binding.goal_id, "run_id": binding.run_id,

                             "session": binding.session, "mission_execution_digest": binding.mission_execution_digest},
                "operations": list(authority.operations), "issued_seq": authority.issued_seq,
                "expires_seq": authority.expires_seq,
            }))
            row = atomic_append(
                Event(1, "scope_supersession_consumed", {
                    "transaction_id": request.transaction_id, "request_ref": request_ref,
                    "authority_token_ref": token_ref,
                    "predecessor_contract_ref": request.predecessor_contract_ref,
                    "successor_contract_ref": request.successor_contract_ref,
                    "required_requalifications": list(obligations),
                    "invalidated_evidence_refs": list(stale), "status": PENDING,
                }),
                transition_id="scope-supersession:" + request.transaction_id,
                expected_head_hash=request.expected_head_hash, domain_tag=self._domain,
            )
            return ConsumptionReceipt(row, obligations, stale)
        except (CapabilityError, ContractValidationError, CasError, LedgerError) as exc:
            raise SupersessionError(str(exc)) from exc

def test_two_independent_sqlite_controllers_consume_once(case):
    c = case
    # Real, distinct SQLite connections and ledger objects. The barrier forces both
    # validators through the precheck; only the storage transaction decides the winner.
    barrier = threading.Barrier(2)
    class RacingLedger(Ledger):
        def append_once_if_new(self, *args, **kwargs):
            barrier.wait(timeout=10)
            return super().append_once_if_new(*args, **kwargs)
    stores = [SqliteLedgerStore(c.path), SqliteLedgerStore(c.path)]
    ledgers = [RacingLedger(s, writer_id="scope-test") for s in stores]
    # Share the actual in-memory CAS to avoid unrelated filesystem-CAS tempfile races.
    copied = CAS(InMemoryCasStore())
    for shard in (c.path.parent / "cas").iterdir():
        for path in shard.iterdir(): copied.put(path.read_bytes())
    c.cas = copied
    authority = token(c)
    def consume(ledger):
        try:
            service(c, ledger=ledger).consume(c.request, authority)
            return "consumed"
        except c.api.SupersessionError:
            return "denied"
    try:
        with ThreadPoolExecutor(2) as pool:
            results = list(pool.map(consume, ledgers))
        assert sorted(results) == ["consumed", "denied"]
        assert len(c.ledger.read()) == 1
        c.ledger.verify_chain()
    finally:
        for ledger in ledgers: ledger.close()
        for store in stores: store.close()

            raise ContractValidationError(f"{label} must be a canonical non-empty string")

        raise ContractValidationError(
            f"base_commit must be a 40-character SHA-1: {contract.base_commit}"

        raise ContractValidationError("context_manifest_hash must be a 64-character SHA-256")
    if not _SHA64.fullmatch(contract.derivation_register_hash):
        raise ContractValidationError("derivation_register_hash must be a 64-character SHA-256")

    for path_str in contract.allowed_paths:

"""Capability broker — the reference monitor (PAEOS-7.6 §7 / 7.5 T1, A-6).

Task B0.8 (PAEOS-8 §10). A `CapabilityToken` is the only authority an agent ever holds; this
module is the kernel component that **mints** them (Ed25519-signed, unforgeable) and
**verifies** them at every privileged call. The `CapabilityToken` *type* is in `kernel/types.py`
(ratified B0.4); this is its behaviour.

The reference monitor enforces the T1/A-6 invariants:
  * **Unforgeable** — the token is an Ed25519 signature over its own `bound_to`, `operations`,
    and TTL. Tampering any field breaks the signature (§3 pins Ed25519; A-5 keeps the signing
    key in the kernel — agents never hold it).
  * **Bound & immutable** — a *genuine* token is usable only for exactly the `(goal_id, run_id,
    stage, role, session)` it was minted for. A BUILDER token cannot be used for a VERIFIER
    action (SI-3, no role relabeling).
  * **TTL'd** — a token is valid only within `[issued_seq, expires_seq]`; stale-token reuse is
    denied (A-6).
  * **Least authority** — any operation not in the explicit `operations` allow-list is denied
    (deny-by-default, FR-4). No self-grant, no escalation.

Minting is kernel-only: an agent has no broker and no key, so it cannot mint or relabel.
"""

from nacl.exceptions import BadSignatureError
from nacl.signing import SigningKey, VerifyKey

from kernel.types import CapabilityBinding, CapabilityToken, GoalId, Role, RunId, StageId, Ts

# ---- errors ---------------------------------------------------------------


class CapabilityError(Exception):
    """Base for capability faults. Every one is a *denial* (deny-by-default, FR-4)."""


class InvalidToken(CapabilityError):
    """The token's signature does not verify — forged or tampered (T1)."""


class ExpiredToken(CapabilityError):
    """The token is outside its TTL window; stale-token reuse denied (A-6)."""


class BindingMismatch(CapabilityError):
    """A genuine token used outside its `bound_to` — wrong goal/run/stage/role/session (SI-3)."""


class OperationNotPermitted(CapabilityError):
    """The requested operation is not in the token's allow-list; escalation denied."""


# ---- canonical signing payload -------------------------------------------


def _payload(
    binding: CapabilityBinding,
    operations: tuple[str, ...],
    issued_seq: Ts,
    expires_seq: Ts,
) -> bytes:
    """Deterministic bytes the signature covers. Any change to any field changes these bytes,
    so a tampered token fails verification."""
    obj = {
        "goal_id": binding.goal_id,
        "run_id": binding.run_id,
        "stage": binding.stage.value,
        "role": binding.role.value,
        "session": binding.session,
        "mission_execution_digest": binding.mission_execution_digest,
        "operations": list(operations),
        "issued_seq": issued_seq,
        "expires_seq": expires_seq,
    }
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode("ascii")

class CapabilityVerifier:
    """Verify capabilities with a public key only; this object cannot mint authority."""

    def __init__(self, verify_key: VerifyKey | bytes) -> None:
        self._verify_key = verify_key if isinstance(verify_key, VerifyKey) else VerifyKey(verify_key)

    def verify(
        self,
        token: CapabilityToken,
        *,
        goal_id: GoalId,
        run_id: RunId,
        stage: StageId,
        role: Role,
        operation: str | None,
        current_seq: Ts,
        session: str | None = None,
        mission_execution_digest: str | None = None,
    ) -> None:
        try:
            self._verify_key.verify(
                _payload(token.bound_to, token.operations, token.issued_seq, token.expires_seq),
                bytes.fromhex(token.token),
            )
        except (BadSignatureError, ValueError) as exc:
            raise InvalidToken(f"token signature does not verify: {exc}") from exc
        if not (token.issued_seq <= current_seq <= token.expires_seq):
            raise ExpiredToken(
                f"current_seq {current_seq} outside TTL "
                f"[{token.issued_seq}, {token.expires_seq}]"
            )
        binding = token.bound_to
        if (
            binding.goal_id != goal_id
            or binding.run_id != run_id
            or binding.stage is not stage
            or binding.role is not role
        ):
            raise BindingMismatch("capability identity differs from its execution request")
        if session is not None and binding.session != session:
            raise BindingMismatch("capability session differs from its execution request")
        if (
            mission_execution_digest is not None
            and binding.mission_execution_digest != mission_execution_digest
        ):
            raise BindingMismatch("capability is not bound to the current mission execution context")
        if operation is not None and operation not in token.operations:
            raise OperationNotPermitted(
                f"operation {operation!r} not in allow-list {token.operations}"
            )


class CapabilityBroker:
    """Mints and verifies capability tokens. Holds the kernel signing key (A-5); an agent has
    neither the broker nor the key, so it can neither mint nor relabel a token."""

    def __init__(self, signing_key: SigningKey | None = None) -> None:
        self._signing_key: SigningKey = (
            signing_key if signing_key is not None else SigningKey.generate()
        )
        self._verify_key: VerifyKey = self._signing_key.verify_key
        self._verifier = CapabilityVerifier(self._verify_key)

    def mint(
        self,
        *,
        goal_id: GoalId,
        run_id: RunId,
        stage: StageId,
        role: Role,
        session: str,
        operations: tuple[str, ...],
        issued_seq: Ts,
        expires_seq: Ts,
        mission_execution_digest: str | None = None,
    ) -> CapabilityToken:
        """Kernel-only: mint a signed token bound to (goal, run, stage, role, session)."""
        if expires_seq < issued_seq:
            raise ValueError(f"expires_seq ({expires_seq}) precedes issued_seq ({issued_seq})")
        binding = CapabilityBinding(
            goal_id=goal_id,
            run_id=run_id,
            stage=stage,
            role=role,
            session=session,
            mission_execution_digest=mission_execution_digest,
        )
        signature = self._signing_key.sign(_payload(binding, operations, issued_seq, expires_seq))
        return CapabilityToken(
            token=signature.signature.hex(),
            bound_to=binding,
            operations=operations,
            issued_seq=issued_seq,
            expires_seq=expires_seq,
        )

    def verify(
        self,
        token: CapabilityToken,
        *,
        goal_id: GoalId,
        run_id: RunId,
        stage: StageId,
        role: Role,
        operation: str,
        current_seq: Ts,
        session: str | None = None,
        mission_execution_digest: str | None = None,
    ) -> None:
        """Reference-monitor check. Raises the specific `CapabilityError` on any failure; returns
        None only if the token is genuine, in-TTL, bound to exactly the expected identity, and
        authorises `operation`. Order: signature → TTL → binding → allow-list."""
        self._verifier.verify(
            token,
            goal_id=goal_id,
            run_id=run_id,
            stage=stage,
            role=role,
            operation=operation,
            current_seq=current_seq,
            session=session,
            mission_execution_digest=mission_execution_digest,
        )

    def verifier(self) -> CapabilityVerifier:
        """Return the public-only verifier for independently checking issued capabilities."""
        return self._verifier

    @property
    def verify_key_hex(self) -> str:
        """The public verify key (hex). Safe to share; the signing key never leaves the broker."""
        return bytes(self._verify_key).hex()
